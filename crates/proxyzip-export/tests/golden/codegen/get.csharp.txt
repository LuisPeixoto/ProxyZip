using var client = new HttpClient();
using var request = new HttpRequestMessage(new HttpMethod("GET"), "https://api.example.com/v1/items?id=2");
request.Headers.TryAddWithoutValidation("Content-Type", "application/json");
request.Headers.TryAddWithoutValidation("Authorization", "Bearer abc");
request.Headers.TryAddWithoutValidation("X-Note", "it's \"quoted\" $5 #{x}");
var response = await client.SendAsync(request);
Console.WriteLine($"{(int)response.StatusCode} {await response.Content.ReadAsStringAsync()}");
